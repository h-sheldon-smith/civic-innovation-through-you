Commands on new computer to run the server:

# Create virtual environment
python -m venv .venv

# Windows:
.venv\Scripts\activate

# WSL/Linux:
source .venv/bin/activate

# Install python packages
pip install -r requirements.txt

# Install app models
python manage.py migrate

# Run server
python manage.py runserver

# Create Super User (SuperUser Account):
python manage.py createsuperuser

# Promote Existing Account to Site Admin (City Administrator Account):
python manage.py promote_to_site_admin username

# Create user
python manage.py shell
from django.contrib.auth import get_user_model; get_user_model().objects.create_user(username='admin1',password='letmein!1!',email='foo@b.com')

# To run unit tests
python -m unittest app.tests.test_something


# DOCKER + EC2 DEPLOYMENT
# DOCKER
# build the docker image
docker build -t city-image .

# runs the image locally, optional
docker run --env-file .env -p 8000:8000 city-image

# log docker into AWS
aws ecr get-login-password --region <your region. i.e. us-east-1> \
| docker login --username AWS --password-stdin <your AWS acct #>.dkr.ecr.<your region>.amazonaws.com

# tag docker image
docker tag <image name i.e. city-image>:latest <aws acct #>.dkr.ecr.<your region>.amazonaws.com/<your ecr repo name>:latest

# push docker image to ecr
docker push <aws acct #>.dkr.ecr.<your region>.amazonaws.com/<your ecr repo name>:latest


# EC2
# Run EC2 instance (note: you have to remove inheritence from the key's security properties)
ssh -i "path-to-vockey\your-vockey-name.pem" ubuntu@<your public ipv4 address>

# Set up and start docker in EC2
sudo apt update
sudo apt install -y docker.io
sudo systemctl enable docker
sudo systemctl start docker

# Install AWS CLI
sudo apt update
sudo apt install -y awscli
aws configure <enter keys, session token, region>

# If EC2 has small memory
sudo fallocate -l 2G /swapfile
sudo chmod 600 /swapfile
sudo mkswap /swapfile
sudo swapon /swapfile

# Launch ecr and pull the docker image
aws ecr get-login-password --region us-east-1 | docker login --username AWS --password-stdin <your ecr uri>
docker pull <your ecr uri>:latest

# Run docker, inject secret keys at runtime
docker run -d -p 8000:8000 \
-e SECRET_KEY='<your secret key>' \
-e ANTHROPIC_KEY='<your secret anthropic key>' \
<aws acct #>.dkr.ecr.<your region>.amazonaws.com/<your ecr repo name>:latest
